```javascript
// Node.js 示例
import fs from 'fs';

const API_URL = 'https://llm.tokencloud.ai/v1/chat/completions';
const API_KEY = process.env.GEMINI_API_KEY;

async function generateImage(inputImagePath, prompt) {
  // 1. 读取并转换图片为 base64
  const imageBuffer = fs.readFileSync(inputImagePath);
  const base64Image = imageBuffer.toString('base64');
  const mimeType = 'image/jpeg';
  const dataUrl = `data:${mimeType};base64,${base64Image}`;

  // 2. 构建请求
  const response = await fetch(API_URL, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${API_KEY}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      model: 'openrouter/google/gemini-3-pro-image-preview',
      messages: [
        {
          role: 'user',
          content: [
            { type: 'image_url', image_url: { url: dataUrl } },
            { type: 'text', text: prompt },
          ],
        },
      ],
      modalities: ['image', 'text'],
    }),
  });

  // 3. 提取生成的图片
  const result = await response.json();
  const generatedImageUrl = result.choices[0].message.images[0].image_url.url;

  // 4. 保存图片
  const base64Data = generatedImageUrl.split(',')[1];
  const buffer = Buffer.from(base64Data, 'base64');
  fs.writeFileSync('output.png', buffer);

  return 'output.png';
}

// 使用
await generateImage('input.jpg', 'Transform this person to have long wavy blonde hair');
```

```javascript

  }
```

### 2. Base64 编码

图片必须转换为 Base64 Data URL 格式：

```
data:<mime-type>;base64,<base64-encoded-data>
```

**支持的格式**：
- `image/jpeg` (.jpg, .jpeg)
- `image/png` (.png)
- `image/webp` (.webp)
- `image/gif` (.gif)

指定输出模态：


## 📝 代码示例

### 示例 1: 基础图片转换 (Node.js)

```javascript
import fs from 'fs';
import path from 'path';

class ImageGenerator {
  constructor(apiKey) {
    this.apiKey = apiKey;
    this.apiUrl = 'https://llm.tokencloud.ai/v1/chat/completions';
    this.model = 'openrouter/google/gemini-3-pro-image-preview';
  }

  // 图片转 Base64
  imageToBase64(imagePath) {
    const imageBuffer = fs.readFileSync(imagePath);
    const base64 = imageBuffer.toString('base64');
    const ext = path.extname(imagePath).toLowerCase();

    const mimeTypes = {
      '.jpg': 'image/jpeg',
      '.jpeg': 'image/jpeg',
      '.png': 'image/png',
      '.webp': 'image/webp',
      '.gif': 'image/gif',
    };

    const mimeType = mimeTypes[ext] || 'image/jpeg';
    return `data:${mimeType};base64,${base64}`;
  }

  // 调用 API
  async transform(imagePath, prompt) {
    const imageDataUrl = this.imageToBase64(imagePath);

    const response = await fetch(this.apiUrl, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.apiKey}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        model: this.model,
        messages: [
          {
            role: 'user',
            content: [
              { type: 'image_url', image_url: { url: imageDataUrl } },
              { type: 'text', text: prompt },
            ],
          },
        ],
        modalities: ['image', 'text'],
      }),
    });

    if (!response.ok) {
      throw new Error(`API Error: ${response.status}`);
    }

    const result = await response.json();
    return result.choices[0].message.images[0].image_url.url;
  }

  // 保存图片
  saveImage(dataUrl, outputPath) {
    const base64Data = dataUrl.split(',')[1];
    const buffer = Buffer.from(base64Data, 'base64');
    fs.writeFileSync(outputPath, buffer);
  }
}

// 使用
const generator = new ImageGenerator(process.env.GEMINI_API_KEY);
const resultUrl = await generator.transform('input.jpg', 'long wavy blonde hair');
generator.saveImage(resultUrl, 'output.png');
```

### 示例 3: 批量处理

```javascript
class BatchImageProcessor {
  constructor(apiKey, concurrency = 3) {
    this.generator = new ImageGenerator(apiKey);
    this.concurrency = concurrency;
  }

  async processBatch(tasks) {
    const results = [];

    // 分批处理
    for (let i = 0; i < tasks.length; i += this.concurrency) {
      const batch = tasks.slice(i, i + this.concurrency);

      const batchPromises = batch.map(async (task) => {
        try {
          const resultUrl = await this.generator.transform(
            task.inputPath,
            task.prompt
          );
          this.generator.saveImage(resultUrl, task.outputPath);
          return { success: true, outputPath: task.outputPath };
        } catch (error) {
          return { success: false, error: error.message };
        }
      });

      const batchResults = await Promise.all(batchPromises);
      results.push(...batchResults);

      // 进度反馈
      console.log(`Processed ${results.length}/${tasks.length}`);
    }

    return results;
  }
}

// 使用
const processor = new BatchImageProcessor(process.env.GEMINI_API_KEY, 3);
const tasks = [
  { inputPath: 'photo1.jpg', prompt: 'long hair', outputPath: 'out1.png' },
  { inputPath: 'photo2.jpg', prompt: 'short hair', outputPath: 'out2.png' },
  { inputPath: 'photo3.jpg', prompt: 'curly hair', outputPath: 'out3.png' },
];

const results = await processor.processBatch(tasks);
console.log(`Success: ${results.filter(r => r.success).length}/${results.length}`);
```

### 示例 5: Python 实现

```python
import base64
import requests
import json
from pathlib import Path

class ImageGenerator:
    def __init__(self, api_key):
        self.api_key = api_key
        self.api_url = 'https://llm.tokencloud.ai/v1/chat/completions'
        self.model = 'openrouter/google/gemini-3-pro-image-preview'

    def image_to_base64(self, image_path):
        """读取图片并转换为 base64"""
        with open(image_path, 'rb') as f:
            image_data = f.read()

        base64_data = base64.b64encode(image_data).decode('utf-8')

        # 检测 MIME 类型
        ext = Path(image_path).suffix.lower()
        mime_types = {
            '.jpg': 'image/jpeg',
            '.jpeg': 'image/jpeg',
            '.png': 'image/png',
            '.webp': 'image/webp',
            '.gif': 'image/gif',
        }
        mime_type = mime_types.get(ext, 'image/jpeg')

        return f'data:{mime_type};base64,{base64_data}'

    def transform(self, image_path, prompt):
        """转换图片"""
        data_url = self.image_to_base64(image_path)

        payload = {
            'model': self.model,
            'messages': [
                {
                    'role': 'user',
                    'content': [
                        {
                            'type': 'image_url',
                            'image_url': {'url': data_url}
                        },
                        {
                            'type': 'text',
                            'text': prompt
                        }
                    ]
                }
            ],
            'modalities': ['image', 'text']
        }

        headers = {
            'Authorization': f'Bearer {self.api_key}',
            'Content-Type': 'application/json'
        }

        response = requests.post(self.api_url, headers=headers, json=payload)
        response.raise_for_status()

        result = response.json()
        return result['choices'][0]['message']['images'][0]['image_url']['url']

    def save_image(self, data_url, output_path):
        """保存图片"""
        # 提取 base64 数据
        base64_data = data_url.split(',')[1]
        image_data = base64.b64decode(base64_data)

        with open(output_path, 'wb') as f:
            f.write(image_data)

# 使用
import os
generator = ImageGenerator(os.environ.get('GEMINI_API_KEY'))
result_url = generator.transform('input.jpg', 'long wavy blonde hair')
generator.save_image(result_url, 'output.png')
```

### 4. 缓存策略

```javascript
class CachedImageGenerator {
  constructor(apiKey, cacheDir = './cache') {
    this.generator = new ImageGenerator(apiKey);
    this.cacheDir = cacheDir;
    fs.mkdirSync(cacheDir, { recursive: true });
  }

  getCacheKey(imagePath, prompt) {
    const crypto = require('crypto');
    return crypto
      .createHash('md5')
      .update(`${imagePath}:${prompt}`)
      .digest('hex');
  }

  async transform(imagePath, prompt) {
    const cacheKey = this.getCacheKey(imagePath, prompt);
    const cachePath = path.join(this.cacheDir, `${cacheKey}.png`);

    // 检查缓存
    if (fs.existsSync(cachePath)) {
      console.log('Cache hit!');
      return this.generator.imageToBase64(cachePath);
    }

    // 调用 API
    const resultUrl = await this.generator.transform(imagePath, prompt);

    // 保存到缓存
    this.generator.saveImage(resultUrl, cachePath);

    return resultUrl;
  }
}
```

### 5. 速率限制

```javascript
class RateLimitedGenerator {
  constructor(apiKey, requestsPerMinute = 20) {
    this.generator = new ImageGenerator(apiKey);
    this.queue = [];
    this.interval = 60000 / requestsPerMinute;
    this.lastRequestTime = 0;
  }

  async transform(imagePath, prompt) {
    const now = Date.now();
    const timeSinceLastRequest = now - this.lastRequestTime;

    if (timeSinceLastRequest < this.interval) {
      const waitTime = this.interval - timeSinceLastRequest;
      await this.sleep(waitTime);
    }

    this.lastRequestTime = Date.now();
    return await this.generator.transform(imagePath, prompt);
  }

  sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}
```

### 常见错误码

| 状态码 | 含义 | 解决方案 |
|--------|------|----------|
| 400 | 请求格式错误 | 检查请求体格式 |
| 401 | API Key 无效 | 验证 API Key |
| 403 | 权限不足 | 检查账户权限 |
| 429 | 请求过于频繁 | 实现速率限制 |
| 500 | 服务器错误 | 稍后重试 |
| 504 | 请求超时 | 增加超时时间或重试 |

### 错误处理示例

```javascript
async function handleApiError(error, context) {
  console.error(`Error in ${context}:`, error);

  if (error.response) {
    const status = error.response.status;
    const data = await error.response.text();

    switch (status) {
      case 400:
        throw new Error(`Bad request: ${data}`);
      case 401:
        throw new Error('Invalid API key. Please check your GEMINI_API_KEY');
      case 403:
        throw new Error('Access forbidden. Check your account permissions');
      case 429:
        throw new Error('Rate limit exceeded. Please wait and retry');
      case 500:
      case 502:
      case 503:
        throw new Error('Server error. Please retry later');
      case 504:
        throw new Error('Request timeout. Try with a smaller image');
      default:
        throw new Error(`API error ${status}: ${data}`);
    }
  }

  throw error;
}
```

### 2. 并发控制

```javascript
class ConcurrentProcessor {
  constructor(apiKey, maxConcurrent = 5) {
    this.generator = new ImageGenerator(apiKey);
    this.maxConcurrent = maxConcurrent;
    this.running = 0;
    this.queue = [];
  }

  async process(task) {
    // 等待可用槽位
    while (this.running >= this.maxConcurrent) {
      await new Promise(resolve => setTimeout(resolve, 100));
    }

    this.running++;

    try {
      const result = await this.generator.transform(task.input, task.prompt);
      return { success: true, result };
    } catch (error) {
      return { success: false, error };
    } finally {
      this.running--;
    }
  }
}
```

### 3. 超时控制

```javascript
async function withTimeout(promise, timeoutMs = 60000) {
  let timeoutId;

  const timeoutPromise = new Promise((_, reject) => {
    timeoutId = setTimeout(() => {
      reject(new Error(`Operation timeout after ${timeoutMs}ms`));
    }, timeoutMs);
  });

  try {
    return await Promise.race([promise, timeoutPromise]);
  } finally {
    clearTimeout(timeoutId);
  }
}
```

```javascript
class MemoryEfficientProcessor {
  async processLargeImage(imagePath, prompt) {
    // 使用流式处理避免一次性加载大图
    const stream = fs.createReadStream(imagePath);
    const chunks = [];

    for await (const chunk of stream) {
      chunks.push(chunk);
    }

    const buffer = Buffer.concat(chunks);
    const base64 = buffer.toString('base64');

    // 清理内存
    chunks.length = 0;

    return this.transform(base64, prompt);
  }
}
```

### 使用统计

```javascript
class AnalyticsWrapper {
  constructor(generator) {
    this.generator = generator;
    this.stats = {
      totalRequests: 0,
      successCount: 0,
      errorCount: 0,
      totalDuration: 0,
    };
  }

  async transform(imagePath, prompt) {
    this.stats.totalRequests++;
    const start = Date.now();

    try {
      const result = await this.generator.transform(imagePath, prompt);
      this.stats.successCount++;
      return result;
    } catch (error) {
      this.stats.errorCount++;
      throw error;
    } finally {
      this.stats.totalDuration += Date.now() - start;
    }
  }

  getStats() {
    return {
      ...this.stats,
      averageDuration: this.stats.totalDuration / this.stats.totalRequests,
      successRate: (this.stats.successCount / this.stats.totalRequests) * 100,
    };
  }
}
```

## ❓ 常见问题

### Q1: 处理时间多久？

**A**: 通常 10-30 秒，取决于：
- 图片大小
- 网络速度
- API 服务器负载

### Q2: 图片大小限制？

**A**:
- 推荐：< 5MB
- 最大：视 API 配额而定
- 建议：压缩后再上传

### Q3: 如何提高生成质量？

**A**:
1. 使用高质量输入图片
2. 写详细的提示词
3. 指定保留原有特征
4. 尝试不同的提示词变体

### Q4: API 配额如何计算？

**A**:
- 按请求次数计费
- 图片大小影响成本
- 查看：https://tokencloud.ai/pricing

### Q5: 支持批量处理吗？

**A**:
- API 本身不支持批量
- 可以在客户端实现并发
- 注意速率限制

### Q6: 如何处理失败？

**A**:
1. 实现重试机制
2. 记录错误日志
3. 降级处理（返回原图或默认图）
4. 用户友好的错误提示

### Q7: 能否离线使用？

**A**:
- 不能，必须联网调用 API
- 可以考虑缓存结果
- 或使用本地模型（需要其他方案）

```javascript
// 复制此代码快速开始
import fs from 'fs';

async function quickStart(imagePath, prompt) {
  const base64 = fs.readFileSync(imagePath).toString('base64');
  const dataUrl = `data:image/jpeg;base64,${base64}`;

  const response = await fetch('https://llm.tokencloud.ai/v1/chat/completions', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${process.env.GEMINI_API_KEY}`,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      model: 'openrouter/google/gemini-3-pro-image-preview',
      messages: [{
        role: 'user',
        content: [
          { type: 'image_url', image_url: { url: dataUrl } },
          { type: 'text', text: prompt },
        ],
      }],
      modalities: ['image', 'text'],
    }),
  });

  const result = await response.json();
  const outputUrl = result.choices[0].message.images[0].image_url.url;

  const outputBuffer = Buffer.from(outputUrl.split(',')[1], 'base64');
  fs.writeFileSync('output.png', outputBuffer);

  console.log('✅ Generated: output.png');
}

quickStart('input.jpg', 'Transform to long wavy blonde hair');
```
